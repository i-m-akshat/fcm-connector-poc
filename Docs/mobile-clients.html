📱 Mobile Client SDK Integration Guide

Complete integration recipes for Swift (iOS), Kotlin (Android), Flutter, and React Native mobile clients connecting to the .NET 8 Push Gateway.

The Client Integration Flow (5 Steps)
  1. Initialize SDK: Boot Firebase in the application entry point.
  2. Request Permissions: Prompt the user for notification authorization.
  3. Register Device: Post the FCM token + appId to /api/devices/register.
  4. Sliding Heartbeat: Start a 45-second timer calling /api/presence/heartbeat.
  5. Handle Payloads: Handle foreground banners, action buttons, and silent background fetches.

🍎 1. Swift (Native iOS)

Include FirebaseCore and FirebaseMessaging via Swift Package Manager or CocoaPods.

AppDelegate.swiftSwift
import UIKit
import FirebaseCore
import FirebaseMessaging
import UserNotifications

@main
class AppDelegate: UIResponder, UIApplicationDelegate, UNUserNotificationCenterDelegate, MessagingDelegate {

    var window: UIWindow?
    var heartbeatTimer: Timer?
    var currentFCMToken: String?
    let currentUserId = "alex@company.com"
    let currentAppId = "CustomerApp" // Matches FIREBASE_APP_CUSTOMERAPP_JSON
    let apiBaseUrl = "https://push-api.company.com/api"

    func application(_ application: UIApplication, didFinishLaunchingWithOptions launchOptions: [UIApplication.LaunchOptionsKey: Any]?) -> Bool {
        // 1. Initialize Firebase
        FirebaseApp.configure()
        Messaging.messaging().delegate = self

        // 2. Request Notification Authorization
        UNUserNotificationCenter.current().delegate = self
        UNUserNotificationCenter.current().requestAuthorization(options: [.alert, .badge, .sound]) { granted, _ in
            if granted {
                DispatchQueue.main.async {
                    application.registerForRemoteNotifications()
                }
            }
        }
        return true
    }

    // 3. Receive FCM Registration Token
    func messaging(_ messaging: Messaging, didReceiveRegistrationToken fcmToken: String?) {
        guard let token = fcmToken else { return }
        self.currentFCMToken = token
        print("✅ FCM Token: \(token)")
        registerDeviceWithBackend(token: token)
    }

    func registerDeviceWithBackend(token: String) {
        guard let url = URL(string: "\(apiBaseUrl)/devices/register") else { return }
        var req = URLRequest(url: url)
        req.httpMethod = "POST"
        req.setValue("application/json", forHTTPHeaderField: "Content-Type")

        let payload: [String: Any] = [
            "userId": currentUserId,
            "appId": currentAppId,
            "deviceToken": token,
            "deviceType": "iOS"
        ]
        req.httpBody = try? JSONSerialization.data(withJSONObject: payload)

        URLSession.shared.dataTask(with: req) { _, response, _ in
            if let http = response as? HTTPURLResponse, http.statusCode == 200 {
                DispatchQueue.main.async {
                    self.startHeartbeatLoop()
                }
            }
        }.resume()
    }

    // 4. Sliding Window 45-Second Heartbeat
    func startHeartbeatLoop() {
        heartbeatTimer?.invalidate()
        heartbeatTimer = Timer.scheduledTimer(withTimeInterval: 45.0, repeats: true) { [weak self] _ in
            self?.sendHeartbeat()
        }
    }

    func sendHeartbeat() {
        guard let token = currentFCMToken,
              let url = URL(string: "\(apiBaseUrl)/presence/heartbeat") else { return }
        var req = URLRequest(url: url)
        req.httpMethod = "POST"
        req.setValue("application/json", forHTTPHeaderField: "Content-Type")
        let payload = ["userId": currentUserId, "deviceId": token]
        req.httpBody = try? JSONSerialization.data(withJSONObject: payload)
        URLSession.shared.dataTask(with: req).resume()
    }

    // 5. Silent Background Fetch Handling
    func application(_ application: UIApplication, didReceiveRemoteNotification userInfo: [AnyHashable : Any],
                     fetchCompletionHandler completionHandler: @escaping (UIBackgroundFetchResult) -> Void) {
        print("🔕 Silent Background Notification: \(userInfo)")
        // Trigger background SQLite / CoreData sync
        completionHandler(.newData)
    }
}

Declare MyFirebaseMessagingService in your AndroidManifest.xml with the com.google.firebase.MESSAGING_EVENT filter.

MyFirebaseMessagingService.ktKotlin
package com.company.app

import com.google.firebase.messaging.FirebaseMessagingService
import com.google.firebase.messaging.RemoteMessage
import okhttp3.*
import okhttp3.MediaType.Companion.toMediaType
import okhttp3.RequestBody.Companion.toRequestBody
import org.json.JSONObject
import java.io.IOException

class MyFirebaseMessagingService : FirebaseMessagingService() {

    private val apiBaseUrl = "https://push-api.company.com/api"
    private val httpClient = OkHttpClient()

    override fun onNewToken(token: String) {
        super.onNewToken(token)
        registerDeviceWithBackend(token)
    }

    override fun onMessageReceived(remoteMessage: RemoteMessage) {
        super.onMessageReceived(remoteMessage)

        // Case 1: Silent Push (Data-Only Payload)
        if (remoteMessage.notification == null && remoteMessage.data.isNotEmpty()) {
            val action = remoteMessage.data["action"]
            // Trigger WorkManager or local Room DB synchronization
            return
        }

        // Case 2: Visible Notification Banner
        remoteMessage.notification?.let {
            showNotification(it.title ?: "Alert", it.body ?: "")
        }
    }

    private fun registerDeviceWithBackend(token: String) {
        val json = JSONObject().apply {
            put("userId", "alex@company.com")
            put("appId", "CustomerApp")
            put("deviceToken", token)
            put("deviceType", "Android")
        }
        val body = json.toString().toRequestBody("application/json".toMediaType())
        val req = Request.Builder().url("$apiBaseUrl/devices/register").post(body).build()
        httpClient.newCall(req).enqueue(object : Callback {
            override fun onFailure(call: Call, e: IOException) {}
            override fun onResponse(call: Call, response: Response) {}
        })
    }

    private fun showNotification(title: String, body: String) {

    }
}

💙 3. Flutter (Cross-Platform)

Using the official firebase_messaging and firebase_core packages.

main.dartDart
import 'dart:async';
import 'dart:convert';
import 'package:flutter/material.dart';
import 'package:firebase_core/firebase_core.dart';
import 'package:firebase_messaging/firebase_messaging.dart';
import 'package:http/http.dart' as http;

const String apiBaseUrl = "https://push-api.company.com/api";
const String userId = "alex@company.com";
const String appId = "CustomerApp";

@pragma('vm:entry-point')
Future _firebaseMessagingBackgroundHandler(RemoteMessage message) async {
  await Firebase.initializeApp();
  if (message.notification == null && message.data.isNotEmpty) {
    print("🔕 Silent Background Push: ${message.data}");
    // Run background local storage sync
  }
}

void main() async {
  WidgetsFlutterBinding.ensureInitialized();
  await Firebase.initializeApp();
  FirebaseMessaging.onBackgroundMessage(_firebaseMessagingBackgroundHandler);
  runApp(const MyApp());
}

class MyApp extends StatefulWidget {
  const MyApp({super.key});
  @override
  State createState() => _MyAppState();
}

class _MyAppState extends State {
  Timer? heartbeatTimer;

  @override
  void initState() {
    super.initState();
    initFCM();
  }

  Future initFCM() async {
    // 1. Request Permission
    NotificationSettings settings = await FirebaseMessaging.instance.requestPermission(
      alert: true, badge: true, sound: true,
    );

    if (settings.authorizationStatus == AuthorizationStatus.authorized) {
      // 2. Obtain Token
      String? token = await FirebaseMessaging.instance.getToken();
      if (token != null) {
        await registerDevice(token);
        startHeartbeatLoop(token);
      }

      // 3. Listen for Token Rotations
      FirebaseMessaging.instance.onTokenRefresh.listen(registerDevice);

      // 4. Handle Foreground Messages
      FirebaseMessaging.onMessage.listen((RemoteMessage message) {
        if (message.notification != null) {
          ScaffoldMessenger.of(context).showSnackBar(
            SnackBar(content: Text("${message.notification!.title}: ${message.notification!.body}")),
          );
        }
      });
    }
  }

  Future registerDevice(String token) async {
    await http.post(
      Uri.parse("$apiBaseUrl/devices/register"),
      headers: {"Content-Type": "application/json"},
      body: jsonEncode({
        "userId": userId,
        "appId": appId,
        "deviceToken": token,
        "deviceType": "FlutterMobile",
      }),
    );
  }

  void startHeartbeatLoop(String token) {
    heartbeatTimer?.cancel();
    heartbeatTimer = Timer.periodic(const Duration(seconds: 45), (timer) async {
      await http.post(
        Uri.parse("$apiBaseUrl/presence/heartbeat"),
        headers: {"Content-Type": "application/json"},
        body: jsonEncode({"userId": userId, "deviceId": token}),
      );
    });
  }

  @override
  void dispose() {
    heartbeatTimer?.cancel();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return const MaterialApp(
      home: Scaffold(body: Center(child: Text("Firebase Client Connected"))),
    );
  }
}

⚛️ 4. React Native

Using @react-native-firebase/app and @react-native-firebase/messaging.

App.jsxJavaScript
import React, { useEffect } from 'react';
import messaging from '@react-native-firebase/messaging';

const API_BASE = 'https://push-api.company.com/api';
const USER_ID = 'alex@company.com';
const APP_ID = 'CustomerApp';

export default function App() {
  useEffect(() => {
    async function setupFCM() {
      // 1. Request Permission
      const authStatus = await messaging().requestPermission();
      const enabled =
        authStatus === messaging.AuthorizationStatus.AUTHORIZED ||
        authStatus === messaging.AuthorizationStatus.PROVISIONAL;

      if (enabled) {
        // 2. Fetch Device Token
        const token = await messaging().getToken();
        await registerDevice(token);

        // 3. Start 45s Heartbeat Loop
        const interval = setInterval(() => {
          sendHeartbeat(token);
        }, 45000);

        return () => clearInterval(interval);
      }
    }

    setupFCM();
  }, []);

  async function registerDevice(token) {
    await fetch(`${API_BASE}/devices/register`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        userId: USER_ID,
        appId: APP_ID,
        deviceToken: token,
        deviceType: 'ReactNative',
      }),
    });
  }

  async function sendHeartbeat(token) {
    await fetch(`${API_BASE}/presence/heartbeat`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ userId: USER_ID, deviceId: token }),
    });
  }

  return null;
}
